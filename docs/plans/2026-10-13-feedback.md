# Feedback Implementation Plan

> Work through the tasks in order. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Anyone signed in reports a problem or suggests an improvement from any page, with an automatic screenshot.
Reporters follow their reports; Admins work through all of them. Design: `docs/design/2026-10-13-feedback.md`.

**Architecture:**
- **Migration 0005** holds the data and the rules:
  - the table `feedback_reports`, with RLS;
  - an evidence trigger;
  - audit through `_audit` in `security definer` code;
  - a private bucket `feedback-screenshots`, with storage policies backed by definer helpers;
  - `set_feedback_status`, `attach_feedback_screenshot`, `feedback_seen`, `feedback_unseen()` and `feedback_counts()`.

  It is re-runnable, and its SQL tests apply it inside a rolled-back transaction, as 0004 does.
- **The app:**
  - A client dialog captures the page with `modern-screenshot`.
  - A server action files the report.
  - The browser uploads the PNG with the reporter's session.
  - A server action links the file through the definer function.
  - `/feedback` lists reports through RLS: Staff see their own, an Admin sees all, with triage controls.

**Tech Stack:** PostgreSQL 17 (Supabase) with RLS and storage, Next.js 16.3.8 (server components, server actions),
React 19.2, Tailwind 4, Zod 4, `lucide-react` 1.48.0, `modern-screenshot` 4.7.0 (new, pinned exactly), Vitest 4,
Playwright 1.63.

## Global Constraints

- Repo is **public**: never commit `.env.local`, real sheets, SO/MO numbers, passwords or keys.
  - Stage files one by one by path. Never use `git add -f`, `git add -A` or `git add .`; git-ignored local scratch
    files must never be committed.
- File names, commit messages and documents are in English. The only Vietnamese text is interface copy in
  `src/messages/vi.ts`, plus the language's name on the VI/EN switch. End-to-end tests may match Vietnamese UI text in
  regular expressions.
- Commits carry no attribution trailer of any kind and no tool names.
- Every user-facing string goes in both dictionaries, with the same keys and placeholders.
- The database is **production**, and its port 5432 is reachable only from the office network.
  - Never run `npm run db:migrate`, `npm test` or `tests/sql` from another network.
  - The SQL tests are written and type-checked; the controller runs them on the office network.
- End-to-end tests run against production. Never start a Playwright run; the controller runs the named specs.
- From a 360 px wide screen (NFR-06): the header, the dialog and the Feedback page fit without horizontal scroll.
  Every control is at least 24 × 24 px.
- Contrast: text at least 4.5:1, non-text at least 3:1, in light and dark mode. The CSP
  (`src/lib/security/csp.ts`) is unchanged.
- Unit tests: `npx vitest run tests/unit`. Also run `npm run lint`, `npm run typecheck` and `npm run build`.

```sql
-- 0005: feedback, a problem report or a suggestion filed from any page (docs/design/2026-10-13-feedback.md).
-- Re-runnable: tests/sql/feedback.test.ts applies this body inside a transaction that is rolled back.
-- Changing policies takes exclusive locks until the transaction ends; give up after 5 s instead of queueing readers.
set local lock_timeout = '5s';

-- ===== Reports =====
create table if not exists public.feedback_reports (
  id              uuid primary key default gen_random_uuid(),
  kind            text not null check (kind in ('broken', 'misread', 'suggestion')),
  impact          text not null check (impact in ('blocking', 'slows_work', 'minor')),
  description     text not null check (length(btrim(description)) >= 5 and length(description) <= 4000),
  page_url        text not null check (length(page_url) between 1 and 500),
  page_route      text not null check (length(page_route) between 1 and 200),
  page_label      text not null check (length(page_label) between 1 and 200),
  screenshot_path text,
  status          text not null default 'new' check (status in ('new', 'looking', 'fixed', 'declined')),
  triage_note     text check (triage_note is null or length(triage_note) <= 2000),
  reporter_id     uuid not null references public.profiles (id),
  created_at      timestamptz not null default now(),
  triaged_by      uuid references public.profiles (id),
  triaged_at      timestamptz,
  constraint feedback_declined_needs_a_reason
    check (status <> 'declined' or length(btrim(coalesce(triage_note, ''))) > 0)
);
create index if not exists feedback_reports_status   on public.feedback_reports (status, created_at desc);
create index if not exists feedback_reports_reporter on public.feedback_reports (reporter_id, created_at desc);
alter table public.feedback_reports enable row level security;

drop policy if exists feedback_file on public.feedback_reports;
create policy feedback_file on public.feedback_reports for insert to authenticated
  with check ((select public.is_allowed_user()) and reporter_id = (select auth.uid()) and status = 'new'
              and screenshot_path is null and triage_note is null and triaged_by is null and triaged_at is null);
drop policy if exists feedback_read on public.feedback_reports;
create policy feedback_read on public.feedback_reports for select to authenticated
  using ((select public.is_allowed_user()) and (reporter_id = (select auth.uid()) or (select public.is_admin())));
-- no update or delete policy: the status moves through set_feedback_status, the picture through attach_feedback_screenshot

revoke all on public.feedback_reports from anon, authenticated;
grant select, insert on public.feedback_reports to authenticated;

-- A filed report is evidence: only the triage columns change, and the picture is attached once.
create or replace function public.trg_feedback_evidence() returns trigger language plpgsql as $$
begin
  if new.id <> old.id or new.kind <> old.kind or new.impact <> old.impact or new.description <> old.description
     or new.page_url <> old.page_url or new.page_route <> old.page_route or new.page_label <> old.page_label
     or new.reporter_id <> old.reporter_id or new.created_at <> old.created_at then
    raise exception 'feedback_is_evidence';
  end if;
  if new.screenshot_path is distinct from old.screenshot_path and old.screenshot_path is not null then
    raise exception 'screenshot_already_attached';
  end if;
  return new;
end $$;
drop trigger if exists feedback_evidence on public.feedback_reports;
create trigger feedback_evidence before update on public.feedback_reports
  for each row execute function public.trg_feedback_evidence();

-- The time a report was filed is the server's, not the client's.
create or replace function public.trg_feedback_stamp() returns trigger language plpgsql as $$
begin
  new.created_at := now();
  return new;
end $$;
drop trigger if exists feedback_stamp on public.feedback_reports;
create trigger feedback_stamp before insert on public.feedback_reports
  for each row execute function public.trg_feedback_stamp();

-- Filing is audited; _audit is closed to clients, so the trigger runs as its owner.
create or replace function public.trg_feedback_filed() returns trigger
language plpgsql security definer set search_path = public, pg_temp as $$
begin
  perform _audit('feedback.file', 'feedback', new.id::text, jsonb_build_object('kind', new.kind, 'impact', new.impact));
  return new;
end $$;
drop trigger if exists feedback_filed on public.feedback_reports;
create trigger feedback_filed after insert on public.feedback_reports
  for each row execute function public.trg_feedback_filed();

-- ===== Triage =====
-- security definer bypasses RLS, so the Admin check is made here. A null note keeps the old one; an empty note clears it.
create or replace function public.set_feedback_status(p_id uuid, p_status text, p_note text default null) returns void
language plpgsql security definer set search_path = public, pg_temp as $$
declare
  v_before text;
  v_old_note text;
  v_note text;
begin
  if not is_admin() then raise exception 'forbidden'; end if;
  if p_status is null or p_status not in ('new', 'looking', 'fixed', 'declined') then raise exception 'invalid_status'; end if;
  if p_note is not null and length(p_note) > 2000 then raise exception 'note_too_long'; end if;
  select status, triage_note into v_before, v_old_note from feedback_reports where id = p_id for update;
  if not found then raise exception 'not_found'; end if;
  v_note := case when p_note is null then v_old_note else nullif(btrim(p_note), '') end;
  if p_status = 'declined' and v_note is null then raise exception 'decline_needs_reason'; end if;
  update feedback_reports
     set status = p_status, triage_note = v_note, triaged_by = auth.uid(), triaged_at = now()
   where id = p_id;
  perform _audit('feedback.status', 'feedback', p_id::text,
                 jsonb_build_object('from', v_before, 'to', p_status, 'note_changed', v_note is distinct from v_old_note));
end $$;

-- ===== The picture =====
create or replace function public.feedback_screenshot_path_valid(p_name text) returns boolean language sql immutable as $$
  select coalesce(p_name ~ '^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}/[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\.png$', false)
$$;

-- A picture goes only into the caller's own report folder, before one is attached, and only one file per folder.
create or replace function public.feedback_screenshot_may_add(p_name text) returns boolean
language plpgsql stable security definer set search_path = public, pg_temp as $$
declare
  v_folder text;
begin
  if not feedback_screenshot_path_valid(p_name) then return false; end if;
  v_folder := split_part(p_name, '/', 1);
  return exists (select 1 from feedback_reports where id = v_folder::uuid and reporter_id = auth.uid() and screenshot_path is null)
     and not exists (select 1 from storage.objects where bucket_id = 'feedback-screenshots' and name like v_folder || '/%');
end $$;

create or replace function public.feedback_screenshot_readable(p_name text) returns boolean
language plpgsql stable security definer set search_path = public, pg_temp as $$
begin
  if not feedback_screenshot_path_valid(p_name) then return false; end if;
  return is_admin() or exists (select 1 from feedback_reports where id = split_part(p_name, '/', 1)::uuid and reporter_id = auth.uid());
end $$;

insert into storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
values ('feedback-screenshots', 'feedback-screenshots', false, 5242880, array['image/png'])
on conflict (id) do update
  set public = false, file_size_limit = excluded.file_size_limit, allowed_mime_types = excluded.allowed_mime_types;

drop policy if exists feedback_shot_add on storage.objects;
create policy feedback_shot_add on storage.objects for insert to authenticated
  with check (bucket_id = 'feedback-screenshots' and (select public.is_allowed_user()) and public.feedback_screenshot_may_add(name));
drop policy if exists feedback_shot_read on storage.objects;
create policy feedback_shot_read on storage.objects for select to authenticated
  using (bucket_id = 'feedback-screenshots' and (select public.is_allowed_user()) and public.feedback_screenshot_readable(name));
-- never updated or deleted by a client

-- Links the uploaded picture to the report. The report has no update policy, so this narrow definer does it.
create or replace function public.attach_feedback_screenshot(p_report_id uuid, p_path text) returns void
language plpgsql security definer set search_path = public, pg_temp as $$
declare
  v_reporter uuid;
  v_existing text;
begin
  if not is_allowed_user() then raise exception 'forbidden'; end if;
  select reporter_id, screenshot_path into v_reporter, v_existing from feedback_reports where id = p_report_id for update;
  if not found or v_reporter is distinct from auth.uid() then raise exception 'not_found'; end if;
  if v_existing is not null then raise exception 'screenshot_already_attached'; end if;
  if not feedback_screenshot_path_valid(p_path) or split_part(p_path, '/', 1) <> p_report_id::text then
    raise exception 'invalid_path';
  end if;
  if not exists (select 1 from storage.objects where bucket_id = 'feedback-screenshots' and name = p_path) then
    raise exception 'file_missing';
  end if;
  update feedback_reports set screenshot_path = p_path where id = p_report_id;
end $$;

-- ===== Counts and the unread signal =====
create or replace function public.feedback_counts() returns table (status text, n bigint)
language sql stable security invoker set search_path = public as $$
  select status, count(*) from feedback_reports group by status
$$;

create table if not exists public.feedback_seen (
  user_id uuid primary key references public.profiles (id),
  seen_at timestamptz not null default now()
);
alter table public.feedback_seen enable row level security;
drop policy if exists feedback_seen_own on public.feedback_seen;
create policy feedback_seen_own on public.feedback_seen for all to authenticated
  using (user_id = (select auth.uid())) with check (user_id = (select auth.uid()));
revoke all on public.feedback_seen from anon, authenticated;
grant select, insert, update on public.feedback_seen to authenticated;

create or replace function public.feedback_seen_at() returns timestamptz
language sql stable security invoker set search_path = public as $$
  select coalesce((select seen_at from feedback_seen where user_id = auth.uid()), '-infinity'::timestamptz)
$$;

-- An Admin: reports still New. Staff: their own reports moved since they last opened the Feedback page.
create or replace function public.feedback_unseen() returns int
language sql stable security invoker set search_path = public as $$
  select case
    when (select is_admin()) then (select count(*) from feedback_reports where status = 'new')
    else (select count(*) from feedback_reports r
           where r.reporter_id = auth.uid() and r.triaged_at is not null and r.triaged_at > feedback_seen_at())
  end::int
$$;

create or replace function public.mark_feedback_seen() returns timestamptz
language plpgsql security invoker set search_path = public as $$
declare
  v_at timestamptz := clock_timestamp(); -- the moment of the visit, not the start of the transaction
begin
  if auth.uid() is null or not is_allowed_user() then raise exception 'forbidden'; end if;
  insert into feedback_seen (user_id, seen_at) values (auth.uid(), v_at)
  on conflict (user_id) do update set seen_at = excluded.seen_at;
  return v_at;
end $$;

-- ===== Privileges =====
revoke all on function
  public.set_feedback_status(uuid, text, text), public.attach_feedback_screenshot(uuid, text),
  public.feedback_screenshot_path_valid(text), public.feedback_screenshot_may_add(text), public.feedback_screenshot_readable(text),
  public.feedback_counts(), public.feedback_seen_at(), public.feedback_unseen(), public.mark_feedback_seen(),
  public.trg_feedback_evidence(), public.trg_feedback_stamp(), public.trg_feedback_filed()
  from public, anon;
grant execute on function
  public.set_feedback_status(uuid, text, text), public.attach_feedback_screenshot(uuid, text),
  public.feedback_screenshot_path_valid(text), public.feedback_screenshot_may_add(text), public.feedback_screenshot_readable(text),
  public.feedback_counts(), public.feedback_seen_at(), public.feedback_unseen(), public.mark_feedback_seen()
  to authenticated;
```

```ts
/** Inserts a report as the table owner (bypassing RLS): arranging data, not exercising the rules. */
export async function insertReportAsOwner(tx: Tx, reporter: TestUser, opts: { status?: string; note?: string; triagedAt?: Date } = {}): Promise<string> {
  const [row] = await tx<{ id: string }[]>`
    insert into public.feedback_reports (kind, impact, description, page_url, page_route, page_label, reporter_id, status, triage_note, triaged_at)
    values ('broken', 'slows_work', 'The list does not load', '/sheets', '/sheets', 'Sheets', ${reporter.id},
            ${opts.status ?? "new"}, ${opts.note ?? null}, ${opts.triagedAt ?? null})
    returning id`;
  return row.id;
}
```

```ts
import { randomUUID } from "node:crypto";
import { afterAll, describe, expect, it } from "vitest";
import {
  actAs, actAsOwner, applyMigration, closeDb, expectError, hasDb, insertObject, insertReportAsOwner, makeUser, rollback, staffEmail, type Tx,
} from "./harness";

afterAll(closeDb);

const MIGRATION = "0005_feedback.sql";
const BUCKET = "feedback-screenshots";

async function world(tx: Tx) {
  await applyMigration(tx, MIGRATION);
  return { a: await makeUser(tx, staffEmail()), b: await makeUser(tx, staffEmail()), admin: await makeUser(tx, staffEmail(), "admin") };
}

/** Files a report with the current role (use after actAs to exercise RLS); `extra` adds or overrides columns. */
const file = (tx: Tx, reporter: string, extra: Record<string, string> = {}) =>
  tx<{ id: string }[]>`insert into public.feedback_reports ${tx({
    kind: "misread", impact: "blocking", description: "The 6.03 was read as 6.08",
    page_url: "/sheets/x", page_route: "/sheets/[id]", page_label: "Editor", reporter_id: reporter, ...extra,
  })} returning id`;

describe.skipIf(!hasDb)("feedback (0005)", () => {
  it("Staff file their own report as new; never with triage fields or for someone else", async () => {
    await rollback(async (tx) => {
      const { a, b } = await world(tx);
      await actAs(tx, a);
      const [ok] = await file(tx, a.id);
      expect(ok.id).toMatch(/^[0-9a-f-]{36}$/);
      await expectError(tx, (sp) => file(sp, b.id), "row-level security");
      await expectError(tx, (sp) => file(sp, a.id, { status: "fixed" }), "row-level security");
      await expectError(tx, (sp) => file(sp, a.id, { triage_note: "x" }), "row-level security");
      await expectError(tx, (sp) => sp`insert into public.feedback_reports (kind, impact, description, page_url, page_route, page_label, reporter_id)
                                        values ('broken', 'minor', '  ab ', '/', '/', 'x', ${a.id})`, "check constraint");
    });
  });

  it("a reporter reads their own reports, an Admin reads every report, nobody else reads them", async () => {
    await rollback(async (tx) => {
      const { a, b, admin } = await world(tx);
      const mine = await insertReportAsOwner(tx, a);
      const read = async () => (await tx<{ id: string }[]>`select id from public.feedback_reports where id = ${mine}`).length;
      await actAs(tx, a);
      expect(await read()).toBe(1);
      await actAsOwner(tx);
      await actAs(tx, b);
      expect(await read()).toBe(0);
      await actAsOwner(tx);
      await actAs(tx, admin);
      expect(await read()).toBe(1);
    });
  });

  it("a filed report is evidence: clients cannot update it, and even the owner role cannot change its content", async () => {
    await rollback(async (tx) => {
      const { a } = await world(tx);
      const id = await insertReportAsOwner(tx, a);
      await actAs(tx, a);
      const changed = await tx`update public.feedback_reports set description = 'Something else entirely' where id = ${id} returning id`;
      expect(changed.length).toBe(0); // no update policy
      await actAsOwner(tx);
      await expectError(tx, (sp) => sp`update public.feedback_reports set description = 'Rewritten afterwards' where id = ${id}`, "feedback_is_evidence");
      await tx`update public.feedback_reports set screenshot_path = ${`${id}/${randomUUID()}.png`} where id = ${id}`;
      await expectError(tx, (sp) => sp`update public.feedback_reports set screenshot_path = null where id = ${id}`, "screenshot_already_attached");
    });
  });

  it("only an Admin moves a report; declining needs a reason; a null note keeps the old one; every move is audited", async () => {
    await rollback(async (tx) => {
      const { a, admin } = await world(tx);
      const id = await insertReportAsOwner(tx, a);
      await actAs(tx, a);
      await expectError(tx, (sp) => sp`select public.set_feedback_status(${id}::uuid, 'fixed', null)`, "forbidden");
      await actAsOwner(tx);
      await actAs(tx, admin);
      await expectError(tx, (sp) => sp`select public.set_feedback_status(${id}::uuid, 'declined', '  ')`, "decline_needs_reason");
      await tx`select public.set_feedback_status(${id}::uuid, 'looking', 'Reproduced on the test sheet')`;
      await tx`select public.set_feedback_status(${id}::uuid, 'declined', null)`; // the kept note is the reason
      await expectError(tx, (sp) => sp`select public.set_feedback_status(${id}::uuid, 'gone', null)`, "invalid_status");
      await actAsOwner(tx);
      const [row] = await tx<{ status: string; triage_note: string; triaged_by: string }[]>`select status, triage_note, triaged_by from public.feedback_reports where id = ${id}`;
      expect(row).toEqual({ status: "declined", triage_note: "Reproduced on the test sheet", triaged_by: admin.id });
      const audit = await tx<{ action: string }[]>`select action from public.audit_log where target_type = 'feedback' and target_id = ${id} order by id`;
      expect(audit.map((r) => r.action)).toEqual(["feedback.file", "feedback.status", "feedback.status"]); // the owner insert is audited too
    });
  });

  it("filing is audited, and the server sets the filing time", async () => {
    await rollback(async (tx) => {
      const { a } = await world(tx);
      await actAs(tx, a);
      const [r] = await file(tx, a.id, { created_at: "2001-01-01T00:00:00Z" });
      await actAsOwner(tx);
      const [row] = await tx<{ old: boolean }[]>`select created_at < now() - interval '1 day' as old from public.feedback_reports where id = ${r.id}`;
      expect(row.old).toBe(false);
      const audit = await tx<{ action: string }[]>`select action from public.audit_log where target_type = 'feedback' and target_id = ${r.id}`;
      expect(audit.map((x) => x.action)).toEqual(["feedback.file"]);
    });
  });

  it("a picture goes into one's own report folder once; it is attached only when it is there; reporter and Admin read it", async () => {
    await rollback(async (tx) => {
      const { a, b, admin } = await world(tx);
      const mine = await insertReportAsOwner(tx, a);
      const theirs = await insertReportAsOwner(tx, b);
      const path = `${mine}/${randomUUID()}.png`;
      await actAs(tx, a);
      await expectError(tx, (sp) => sp`select public.attach_feedback_screenshot(${mine}::uuid, ${path})`, "file_missing");
      await expectError(tx, (sp) => sp`insert into storage.objects (bucket_id, name, owner, owner_id) values (${BUCKET}, ${`${theirs}/${randomUUID()}.png`}, ${a.id}, ${a.id})`, "row-level security");
      await expectError(tx, (sp) => sp`insert into storage.objects (bucket_id, name, owner, owner_id) values (${BUCKET}, ${`${mine}/not-a-uuid.png`}, ${a.id}, ${a.id})`, "row-level security");
      await tx`insert into storage.objects (bucket_id, name, owner, owner_id) values (${BUCKET}, ${path}, ${a.id}, ${a.id})`;
      await expectError(tx, (sp) => sp`insert into storage.objects (bucket_id, name, owner, owner_id) values (${BUCKET}, ${`${mine}/${randomUUID()}.png`}, ${a.id}, ${a.id})`, "row-level security");
      await expectError(tx, (sp) => sp`select public.attach_feedback_screenshot(${theirs}::uuid, ${path})`, "not_found");
      await tx`select public.attach_feedback_screenshot(${mine}::uuid, ${path})`;
      await expectError(tx, (sp) => sp`select public.attach_feedback_screenshot(${mine}::uuid, ${path})`, "screenshot_already_attached");
      const seen = async () => (await tx`select 1 from storage.objects where bucket_id = ${BUCKET} and name = ${path}`).length;
      expect(await seen()).toBe(1);
      await actAsOwner(tx);
      await actAs(tx, b);
      expect(await seen()).toBe(0);
      await actAsOwner(tx);
      await actAs(tx, admin);
      expect(await seen()).toBe(1);
    });
  });

  it("the unread count: New reports for an Admin, one's own moved reports for Staff, cleared by marking seen", async () => {
```

```ts
export async function captureScreen(): Promise<Blob | null> {
```
